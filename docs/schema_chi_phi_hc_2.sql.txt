-- 1. DANH MỤC KHOẢN MỤC PHÍ (thay thế DM_CPHC dạng Google Sheet cũ)
create table if not exists dm_kmp (
  id uuid primary key default gen_random_uuid(),
  ma_b7 text not null,
  ma_b10 text,
  nhom_chi_phi text not null,
  dien_giai text,
  trong_yeu boolean not null default false,
  active boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
comment on table dm_kmp is 'Danh mục Khoản mục phí: Mã B7/B10, Nhóm chi phí — dùng để phân bổ DNTT';

-- 2. DANH MỤC BỘ PHẬN CẤP 1 (dòng kinh doanh / nghiệp vụ)
create table if not exists dm_bo_phan_cap1 (
  id uuid primary key default gen_random_uuid(),
  ma text not null unique,
  ten text not null,
  yeu_cau_cap2 boolean not null default false, -- true = bắt buộc chọn thương hiệu (Cấp 2)
  thu_tu int not null default 0,
  active boolean not null default true
);
comment on table dm_bo_phan_cap1 is 'Cấp 1: KD Xe | KD DV | KD PT-VT | Nghiệp vụ Quản trị chung | Dùng chung & Khác...';

-- 3. DANH MỤC BỘ PHẬN CẤP 2 (thương hiệu)
create table if not exists dm_bo_phan_cap2 (
  id uuid primary key default gen_random_uuid(),
  ma text not null unique,
  ten text not null, -- KIA, MAZDA, PEUGEOT, BMW, BMW MOTORRAD, RAM, JEEP...
  thu_tu int not null default 0,
  active boolean not null default true
);
comment on table dm_bo_phan_cap2 is 'Cấp 2: Thương hiệu — chỉ dùng khi dm_bo_phan_cap1.yeu_cau_cap2 = true. Là chiều ĐỘC LẬP với Cấp 1 (không phải cây cha-con cứng) — cùng 1 thương hiệu (vd Kia) có thể xuất hiện ở nhiều Cấp 1 khác nhau (Kinh doanh Kia, Dịch vụ Kia, Xưởng Sửa chữa Kia).';

-- 4. GIẤY ĐỀ NGHỊ THANH TOÁN (header — khớp mẫu MAU_DNTT.md)
create table if not exists dntt (
  id uuid primary key default gen_random_uuid(),
  so_dntt text unique,                          -- mã tự sinh, ví dụ DNTT-2026-000123
  ngay_lap date not null default current_date,

  -- Thông tin pháp nhân/đơn vị: THAM CHIẾU tới bảng Đơn vị/Pháp nhân/Showroom
  -- đã có sẵn trong QTVP-ASDS (Antigravity xác nhận đúng tên bảng/cột thật khi audit)
  id_phap_nhan uuid,        -- FK tới bảng pháp nhân hiện có (dùng để lấy Tên/Địa chỉ/MST tự nhảy)
  id_don_vi uuid,           -- FK tới bảng Đơn vị/Showroom hiện có (theo bộ lọc Đơn vị bên ngoài)

  nguoi_de_nghi text not null,
  bo_phan_hien_thi text,    -- Ô "Bộ phận" gõ tay tự do, CHỈ để in trên form — KHÔNG liên quan Cấp 1/Cấp 2
  noi_dung_thanh_toan text,

  tong_so_tien numeric(18,2) not null default 0,
  so_tien_bang_chu text,

  hinh_thuc_thanh_toan text not null default 'Chuyển khoản'
    check (hinh_thuc_thanh_toan in ('Chuyển khoản','Tiền mặt')),
  ten_tai_khoan text,
  so_tai_khoan text,
  ten_ngan_hang text,
  chi_nhanh_ngan_hang text,

  trang_thai text not null default 'Nháp'
    check (trang_thai in ('Nháp','Chờ duyệt','Đã duyệt','Đã thanh toán','Từ chối')),

  created_by uuid references auth.users(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
comment on table dntt is 'Giấy Đề nghị Thanh toán — header, khớp mẫu MAU_DNTT.md';

-- 5. CHI TIẾT NỘI DUNG THANH TOÁN (bảng STT | Nội dung | Số tiền trên form)
create table if not exists dntt_chi_tiet (
  id uuid primary key default gen_random_uuid(),
  dntt_id uuid not null references dntt(id) on delete cascade,
  stt int not null,
  noi_dung text not null,
  so_tien numeric(18,2) not null default 0
);
comment on table dntt_chi_tiet is 'Các dòng nội dung thanh toán hiển thị trên giấy DNTT (không phải phân bổ nội bộ)';

-- 6. DÒNG PHÂN BỔ NỘI BỘ (Khoản mục phí + Bộ phận + Kỳ)
-- QUAN TRỌNG: gắn theo TỪNG DÒNG "dntt_chi_tiet" cụ thể (không gắn chung cho cả DNTT),
-- vì thực tế 1 dòng nội dung thanh toán (vd "Tiền điện Tổ hợp Showroom") được chia nhỏ
-- thành nhiều dòng phân bổ con (Kinh doanh Kia, Dịch vụ Mazda...), và các dòng con này
-- PHẢI được in ra ngay trên file DNTT xuất, ngay dưới dòng nội dung cha của nó.
create table if not exists dntt_phan_bo (
  id uuid primary key default gen_random_uuid(),
  dntt_id uuid not null references dntt(id) on delete cascade,          -- để truy vấn nhanh theo cả DNTT
  dntt_chi_tiet_id uuid not null references dntt_chi_tiet(id) on delete cascade, -- dòng nội dung cha cụ thể

  -- Hỗ trợ CẢ 2 cách nhập: theo % hoặc số tiền cố định (tùy đơn vị/người dùng chọn)
  kieu_nhap text not null default 'SO_TIEN' check (kieu_nhap in ('PHAN_TRAM','SO_TIEN')),
  phan_tram numeric(5,2),              -- dùng khi kieu_nhap = 'PHAN_TRAM' (0.00 - 100.00)
  so_tien numeric(18,2) not null check (so_tien > 0),  -- luôn có giá trị cuối cùng (tự tính từ % hoặc nhập tay)

  id_kmp uuid not null references dm_kmp(id),
  thang int not null check (thang between 1 and 12),
  nam int not null check (nam between 2000 and 2100),

  id_bo_phan_cap1 uuid not null references dm_bo_phan_cap1(id),
  id_bo_phan_cap2 uuid references dm_bo_phan_cap2(id), -- nullable, chỉ khi cap1.yeu_cau_cap2 = true

  thu_tu int not null default 0,  -- thứ tự hiển thị dòng con khi in ra file
  ghi_chu text,
  created_at timestamptz not null default now()
);
comment on table dntt_phan_bo is 'Mỗi dòng = 1 phần tiền của 1 dòng nội dung thanh toán cụ thể (dntt_chi_tiet), gắn Khoản mục phí + Kỳ + Bộ phận. Tổng các dòng phân bổ của 1 dntt_chi_tiet_id phải bằng đúng so_tien của dòng đó (kiểm tra ở tầng ứng dụng trước khi cho tải file) — các dòng này được in trực tiếp lên file DNTT xuất, ngay dưới dòng nội dung cha.';

-- Index phục vụ truy vấn báo cáo
create index if not exists idx_dntt_phan_bo_dntt on dntt_phan_bo(dntt_id);
create index if not exists idx_dntt_phan_bo_chitiet on dntt_phan_bo(dntt_chi_tiet_id);
create index if not exists idx_dntt_phan_bo_ky on dntt_phan_bo(nam, thang);
create index if not exists idx_dntt_phan_bo_kmp on dntt_phan_bo(id_kmp);
create index if not exists idx_dntt_status on dntt(trang_thai);

-- Kiểm tra tổng phân bổ khớp tổng của TỪNG DÒNG NỘI DUNG (không phải khớp cả DNTT gộp chung)
create or replace view v_dntt_chi_tiet_check_tong as
select
  ct.id as dntt_chi_tiet_id,
  ct.dntt_id,
  ct.noi_dung,
  ct.so_tien as so_tien_dong_goc,
  coalesce(sum(pb.so_tien), 0) as tong_da_phan_bo,
  ct.so_tien - coalesce(sum(pb.so_tien), 0) as chenh_lech
from dntt_chi_tiet ct
left join dntt_phan_bo pb on pb.dntt_chi_tiet_id = ct.id
group by ct.id, ct.dntt_id, ct.noi_dung, ct.so_tien;
comment on view v_dntt_chi_tiet_check_tong is 'Dùng để kiểm tra nhanh dòng nội dung nào chưa phân bổ đủ 100% số tiền của chính nó';
